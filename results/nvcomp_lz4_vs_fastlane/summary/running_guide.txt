Paste this somewhere as your **FastLanes-GPU command checklist**.

```bash
# ============================================================
# FastLanes-GPU benchmark commands
# Run from repo root:
# ~/gpu_benchmark_clean
# ============================================================

# ----------------------------
# 0) Build FastLanes-GPU tools
# ----------------------------
cmake -S external -B build \
  -DCMAKE_PREFIX_PATH="$PWD/external/local" \
  -Dnvcomp_DIR=/home/riaz/nvcomp_env/lib/python3.12/site-packages/nvidia/libnvcomp/lib64/cmake/nvcomp

cmake --build build --target fastlanes_gpu_compress -j$(nproc)
cmake --build build --target fastlanes_gpu_decompress -j$(nproc)
```

## 1) Original SF10 quantity layout

```bash
# Create FastLanes input folder
mkdir -p results/baseline_fastlanes_gpu/quantity_sf10_original

# Link original SF10 quantity column as raw.dat
ln -sf "$PWD/data/tpch_columnar_sf10/quantity_sf10.bin" \
  results/baseline_fastlanes_gpu/quantity_sf10_original/raw.dat

# Remove old FastLanes output for clean run
rm -rf results/baseline_fastlanes_gpu/quantity_sf10_original/fastlanes_gpu

# Compress
bash external/scripts/fastlanes_gpu/compress.sh \
  results/baseline_fastlanes_gpu/quantity_sf10_original

# Decompress + benchmark
bash external/scripts/fastlanes_gpu/decompress.sh \
  results/baseline_fastlanes_gpu/quantity_sf10_original

# Show result
cat results/baseline_fastlanes_gpu/quantity_sf10_original/fastlanes_gpu/decompress_benchmark.txt

# Optional: remove decompressed output to save disk
rm -f results/baseline_fastlanes_gpu/quantity_sf10_original/fastlanes_gpu/decompressed.dat
```

## Useful result-collection commands

```bash
# Show all saved FastLanes benchmark files
find results/baseline_fastlanes_gpu -name "decompress_benchmark*" -type f -ls

# Print all saved FastLanes benchmark results
for f in $(find results/baseline_fastlanes_gpu -name "decompress_benchmark*" -type f | sort); do
  echo "===== $f ====="
  cat "$f"
  echo
done

# Check disk usage
df -h .
du -h --max-depth=2 results/baseline_fastlanes_gpu | sort -h | tail -30
```
